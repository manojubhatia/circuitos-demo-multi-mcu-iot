cmake_minimum_required(VERSION 3.16)
project(theSpeaker_(3_boards)_desktop_sim C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
add_compile_definitions(DESKTOP_SIM=1 CONFIG_LOG_DEFAULT_LEVEL=4)

add_library(platform_sim STATIC platform/osal_posix.c platform/hal_sim_spi.c platform/hal_sim_gpio.c)
target_include_directories(platform_sim PUBLIC platform/)
target_link_libraries(platform_sim PUBLIC pthread)

file(GLOB_RECURSE S_thespeaker_C "${CMAKE_SOURCE_DIR}/../device_thespeaker/main/*.c")
file(GLOB_RECURSE S_thespeaker_CPP "${CMAKE_SOURCE_DIR}/../device_thespeaker/main/*.cpp")
list(FILTER S_thespeaker_C EXCLUDE REGEX "\\.baselines/")
list(FILTER S_thespeaker_CPP EXCLUDE REGEX "\\.baselines/")
list(FILTER S_thespeaker_C EXCLUDE REGEX "osal_task\\.c|osal_queue\\.c|osal_sync\\.c|osal\\.c")
list(FILTER S_thespeaker_C EXCLUDE REGEX "hal_gpio\\.c|hal_adc\\.c|hal_spi\\.c|hal_pwm\\.c|hal_i2c\\.c")
list(FILTER S_thespeaker_C EXCLUDE REGEX "_esp\\.c")
add_executable(sim_thespeaker ${S_thespeaker_C} ${S_thespeaker_CPP} mcu_thespeaker/sim_sensors.cpp sim_main.c)
target_include_directories(sim_thespeaker PRIVATE platform/ ${CMAKE_SOURCE_DIR}/../device_thespeaker/main/ ${CMAKE_SOURCE_DIR}/../device_thespeaker/main/include/ ${CMAKE_SOURCE_DIR}/../device_thespeaker/main/base/ ${CMAKE_SOURCE_DIR}/../device_thespeaker/main/sensors/ ${CMAKE_SOURCE_DIR}/../device_thespeaker/main/actuators/ ${CMAKE_SOURCE_DIR}/../device_thespeaker/main/osal/ ${CMAKE_SOURCE_DIR}/../device_thespeaker/main/hal/ ${CMAKE_SOURCE_DIR}/../device_thespeaker/main/middleware/ ${CMAKE_SOURCE_DIR}/../device_thespeaker/main/hooks/ ${CMAKE_SOURCE_DIR}/../device_thespeaker/main/periphery_state/ ${CMAKE_SOURCE_DIR}/../device_thespeaker/main/src/ mcu_thespeaker/)
target_link_libraries(sim_thespeaker PRIVATE platform_sim)
target_compile_definitions(sim_thespeaker PRIVATE SIM_MCU_NAME="theSpeaker" SIM_SPI_PORT=9200)
